import Changelog, { getHeadings as changelogHeadings } from '../../../../CHANGELOG.md';

export const versions = changelogHeadings().filter((h) => h.depth === 2 && /^\d+\.\d+\.\d+ \(/.test(h.text));
export const latest = versions[0];

  are also its <a href="https://github.com/barca-orc/barca/releases">GitHub release</a>; changes
  merged since the latest release are in
  [`CHANGELOG.md`](https://github.com/barca-orc/barca/blob/main/CHANGELOG.md) under Pending.
</div>
